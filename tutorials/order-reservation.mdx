## Implement the reservation Component

Create a TypeScript Component named **Reserve order balance** with a visible element named **Response**. Paste the complete element body.

```ts
type Order = {
  orderId: string;
  accountId: string;
  amountMinor: string;
  currency: string;
  status: string;
};

type Balance = {
  balanceId: string;
  accountId: string;
  currency: string;
  availableMinor: string;
  reservedMinor: string;
};

type Reservation = {
  transactionId: string;
  orderId: string;
  operationId: string;
  status: string;
  settlementJobId?: string | null;
};

const { orderId, requestKey } = api.input()?.body as {
  orderId: string;
  requestKey: string;
};

const operationId = api.getOperationId();
const principal = api.currentUser();
const gateKey = `order-reservation:${requestKey}`;
const settleOrderTemplateId = String(
  api.input()?.templateInputs?.settleOrderTemplateId,
);

if (!util.isUUID(orderId)) {
  api.throw(422, { message: 'A valid order ID is required', operationId });
}
if (!util.isUUID(settleOrderTemplateId)) {
  api.throw(500, {
    message: 'Settlement Job template is not configured',
    operationId,
  });
}

const submitSettlement = async (reservation: Reservation): Promise<string | null> => {
  if (reservation.orderId !== orderId) {
    api.throw(409, { message: 'Request key belongs to another order', operationId });
  }
  if (reservation.settlementJobId || reservation.status !== 'PENDING_SETTLEMENT') {
    return reservation.settlementJobId ?? null;
  }

  const candidateJobId = await api.triggerJob(settleOrderTemplateId, {
    transactionId: reservation.transactionId,
  });

  // Keep the first recorded receipt if two requests both reach submission.
  return api.transactionDatabase(async (tx) => {
    const filter = { field: 'transactionId', op: 'eq' as const, value: reservation.transactionId };
    const page = await tx.getDatabaseData<Reservation>(
      'transactions', { filter, take: 1 }, { lock: 'update' },
    );
    const saved = page.data[0];
    if (!saved) throw new Error('Reservation is unavailable');
    if (saved.settlementJobId) return saved.settlementJobId;
    await tx.updateDatabaseData(
      'transactions',
      { transactionId: reservation.transactionId },
      { settlementJobId: candidateJobId },
    );
    return candidateJobId;
  });
};

const acquired = await api.acquireIdempotencyKey(gateKey, 86_400, {
  operationId,
  orderId,
  state: 'PROCESSING',
});

if (!acquired) {
  const existing = await api.getDatabaseData<Reservation>('transactions', {
    filter: { field: 'requestKey', op: 'eq', value: requestKey },
    take: 1,
  });

  if (existing.results === 1) {
    const reservation = existing.data[0];
    return {
      operationId,
      originalOperationId: reservation.operationId,
      duplicate: true,
      transactionId: reservation.transactionId,
      jobId: await submitSettlement(reservation),
      status: reservation.status,
    };
  }

  api.throw(409, {
    message:
      'A request with this idempotency key is already processing',
    operationId,
  });
}

let reservation;

try {
  reservation = await api.transactionDatabase(
    async (tx) => {
      const existingTransactions = await tx.getDatabaseData<Reservation>(
        'transactions',
        {
          filter: { field: 'requestKey', op: 'eq', value: requestKey },
          take: 1,
        },
        { lock: 'update' },
      );

      if (existingTransactions.results === 1) {
        return {
          ...existingTransactions.data[0],
          duplicate: true,
        };
      }

      const orders = await tx.getDatabaseData<Order>(
        'orders',
        {
          filter: { field: 'orderId', op: 'eq', value: orderId },
          take: 1,
        },
        { lock: 'update' },
      );

      if (orders.results !== 1) {
        api.throw(404, { message: 'Order not found', operationId });
      }

      const order = orders.data[0];

      if (order.status !== 'PENDING') {
        api.throw(409, {
          message: 'Order is not eligible for reservation',
          operationId,
          currentStatus: order.status,
        });
      }

      const balances = await tx.getDatabaseData<Balance>(
        'balances',
        {
          filter: {
            and: [
              { field: 'accountId', op: 'eq', value: order.accountId },
              { field: 'currency', op: 'eq', value: order.currency },
            ],
          },
          take: 1,
        },
        { lock: 'update' },
      );

      if (balances.results !== 1) {
        api.throw(409, {
          message: 'Balance is unavailable',
          operationId,
        });
      }

      const balance = balances.data[0];
      const amountMinor = BigInt(order.amountMinor);
      const availableMinor = BigInt(balance.availableMinor);
      const reservedMinor = BigInt(balance.reservedMinor);

      if (amountMinor <= 0n || availableMinor < amountMinor) {
        api.throw(409, {
          message: 'Insufficient available balance',
          operationId,
        });
      }

      const transactionId = util.randomUUID();

      await tx.updateDatabaseData(
        'balances',
        { balanceId: balance.balanceId },
        {
          availableMinor: (availableMinor - amountMinor).toString(),
          reservedMinor: (reservedMinor + amountMinor).toString(),
          lastOperationId: operationId,
        },
      );

      await tx.updateDatabaseData(
        'orders',
        { orderId: order.orderId, status: 'PENDING' },
        {
          status: 'RESERVED',
          reservationTransactionId: transactionId,
          lastOperationId: operationId,
        },
      );

      await tx.insertDatabaseData('transactions', [
        {
          transactionId,
          requestKey,
          orderId: order.orderId,
          accountId: order.accountId,
          type: 'RESERVATION',
          amountMinor: amountMinor.toString(),
          currency: order.currency,
          status: 'PENDING_SETTLEMENT',
          operationId,
          requestedBy: principal.userId,
        },
      ]);

      return {
        duplicate: false,
        transactionId,
        orderId,
        operationId,
        status: 'PENDING_SETTLEMENT',
        settlementJobId: null,
      };
    },
    {
      isolationLevel: 'SERIALIZABLE',
      timeoutMs: 15_000,
    },
  );
} catch (error) {
  await api.releaseIdempotencyKey(gateKey);
  throw error;
}

const jobId = await submitSettlement(reservation);

api.log(
  {
    message: 'Order balance reserved',
    args: {
      operationId,
      orderId,
      transactionId: reservation.transactionId,
      jobId,
      requestedBy: principal.userId,
    },
  },
  'INFO',
);

return {
  operationId,
  transactionId: reservation.transactionId,
  jobId,
  duplicate: reservation.duplicate,
  originalOperationId: reservation.duplicate ? reservation.operationId : undefined,
  status: reservation.status,
};
```

<Warning>
  `api.transactionDatabase()` commits when its callback resolves and rolls back when it throws. Await each `tx.*` call sequentially; parallel transaction operations are rejected. The default timeout is 15 seconds, and explicit `timeoutMs` values are limited to 1–30 seconds.
</Warning>

The idempotency guard and unique `transactions.requestKey` prevent a second reservation. The ledger also keeps the first recorded settlement Job ID. If submission or receipt persistence fails after the reservation commits, repeating the same order ID and request key resumes the handoff. Inspect Job History before repeating an ambiguous submission. A lost receipt can lead to a second Job for the same transaction, so both runs must use the same provider idempotency key and the worker must skip the provider once settlement is recorded.

## Configure the Endpoint

Open **Endpoints**, create the definition inactive, and configure it before activation.

| UI field | Value for this guide | Reason |
| --- | --- | --- |
| Name | `Reserve order balance` | Searchable operator name. |
| Method | `POST` | The command changes state. |
| Path | `/orders/reserve` | Stable public route. |
| Component | `Reserve order balance` | Selects the reviewed logic. |
| Component Version | Exact reviewed version | Prevents an unrelated activation changing the contract. |
| Runtime input | `settleOrderTemplateId` | Keeps deployment configuration outside caller-controlled input. |
| Request guard | `reserveOrderRequest` | Rejects malformed input before execution. |
| Validation status | `422` | Separates invalid input from business conflicts. |
| Success status | `200` | A successful response reports the committed reservation state and the known Job receipt. |
| Result element | **Response** (`popResult`) | Returns the element's JSON directly. |
| Hide validation details | On for public clients | Avoids exposing internal schema details. |
| Hide request | On | Reduces retained sensitive payload data. |
| Hide response | Off or policy-driven | Keeps the returned IDs available to authorized support. |

Keep the definition inactive while setting up [the settlement Job template](/tutorials/order-settlement). Then paste `{"settleOrderTemplateId":"<job-template-id>"}` into **Runtime input**, activate the Component and Endpoint in a test instance, and send the accepted and rejected requests there. Inspect the generated Endpoint OpenAPI contract before production activation.

### Request

Use a seeded test order ID. Set `REVO_ENDPOINT_URL` to the instance's [Endpoint runtime base URL](/developers/endpoint-runtime) and `REVO_SERVICE_ACCOUNT_KEY` to the test caller credential.

```bash
curl --request POST \
  --url "$REVO_ENDPOINT_URL/orders/reserve" \
  --header "Authorization: Bearer $REVO_SERVICE_ACCOUNT_KEY" \
  --header "Content-Type: application/json" \
  --data '{
    "orderId": "0195af19-9b41-7a7c-8c5c-35ab875e165c",
    "requestKey": "checkout-0195af19-9b41-7a7c-8c5c-35ab875e165c-v1"
  }'
```

### Response

```json
{
  "operationId": "0195af23-78d0-7c6c-b444-82238598e122",
  "transactionId": "0195af23-7a2e-7567-a9a2-8ab735cf3788",
  "jobId": "0195af23-7b7a-7d17-ae86-c4b16d8e47ec",
  "duplicate": false,
  "status": "PENDING_SETTLEMENT"
}
```

Also retain the `x-revo-oid` response header in the calling system. It is the fastest starting point for a cross-service trace when a client did not preserve the JSON response.

Send the request again with an empty JSON body to confirm `422`. The guard rejects that request before the Component can reserve funds or start a Job.
